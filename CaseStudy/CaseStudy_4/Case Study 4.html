// Case Study: ATM Card PIN Verification using JavaScript // Functions Case Study // Name: Insiya Hussain // PRN: 24070521156 // Global Scope: ATM System Configuration const BANK_NAME = "Global Secure Bank"; const CORRECT_PIN = "1234"; const MAX_ATTEMPTS = 3; let attempts = 0; function verifyPin(enteredPin) { return enteredPin === CORRECT_PIN; } const validatePinFormat = function(pin) { return /^\d{4}$/.test(pin); }; const generateMessage = (isVerified) => { if (isVerified) { return "PIN verified successfully! Welcome to " + BANK_NAME + "."; } else { return "Incorrect PIN. Please try again."; } }; function processATMVerification(enteredPin) { console.log("\n------------------------------------"); console.log(" ATM PIN VERIFICATION"); console.log("------------------------------------"); console.log("Bank: " + BANK_NAME); if (!validatePinFormat(enteredPin)) { console.log("Invalid PIN. Please enter exactly 4 digits."); return false; } attempts++; const verified = verifyPin(enteredPin); const message = generateMessage(verified); console.log(message); if (!verified) { console.log("Attempts remaining: " + (MAX_ATTEMPTS - attempts)); } return verified; } const readline = require("readline"); const rl = readline.createInterface({ input: process.stdin, output: process.stdout }); function askForPin() { if (attempts >= MAX_ATTEMPTS) { console.log("\nAccount locked. Maximum attempts exceeded."); rl.close(); return; } rl.question("\nEnter your 4-digit ATM PIN: ", function(enteredPin) { const verified = processATMVerification(enteredPin); if (verified) { console.log("\nTransaction can proceed."); rl.close(); } else if (attempts >= MAX_ATTEMPTS) { console.log("\nAccount locked. Maximum attempts exceeded."); rl.close(); } else { askForPin(); } }); } console.log("\n===== WELCOME TO ATM ====="); askForPin();
